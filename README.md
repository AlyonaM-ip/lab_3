### 1. Исходный код clone_thread.c и pthread-версии

#### 1.1 Код реализации ручного потока через clone() с комментариями флагов
```c
#define _GNU_SOURCE
#include <sched.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/syscall.h>

#define STACK_SIZE (1024 * 1024)
static volatile long shared_counter = 0;

static int thread_func(void *arg) {
    long iterations = (long)arg;
    for (long i = 0; i < iterations; i++) {
        shared_counter++; 
    }
    printf("[CHILD] Запущен: PID=%ld, TID=%ld\n", (long)getpid(), (long)syscall(SYS_gettid));
    printf("[CHILD] Адрес разделяемой переменной: %p\n", (void *)&shared_counter);
    return 0;
}

int main(void) {
    printf("[MAIN] Стартовал: PID=%ld, TID=%ld\n", (long)getpid(), (long)syscall(SYS_gettid));
    
    // Выделение пользовательского стека (Шаг 1.1)
    char *stack = malloc(STACK_SIZE);
    if (!stack) { perror("malloc"); return EXIT_FAILURE; }
    char *stack_top = stack + STACK_SIZE; // На x86-64 стек растет сверху вниз

    // Вызов clone с побитовым набором флагов потока (Шаг 1.2)
    pid_t tid = clone(thread_func, stack_top,
                      CLONE_VM      | // Общее виртуальное адресное пространство
                      CLONE_FS      | // Общая информация о файловой системе (cwd, umask)
                      CLONE_FILES   | // Общая таблица файловых дескрипторов
                      CLONE_SIGHAND | // Общая таблица обработчиков сигналов
                      CLONE_THREAD  | // Включение в одну Thread Group (совпадение PID)
                      SIGCHLD, 
                      (void*)10000);

    if (tid == -1) { perror("clone"); free(stack); return EXIT_FAILURE; }

    sleep(1); 
    printf("[MAIN] Финальное значение счётчика: %ld\n", shared_counter);
    free(stack);
    return EXIT_SUCCESS;
}
```

#### 1.2 Эквивалентная потоковая версия на POSIX Threads (pthread_thread.c)
```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <unistd.h>
#include <sys/syscall.h>

static volatile long shared_counter = 0;

static void *worker(void *arg) {
    long iterations = (long)arg;
    for (long i = 0; i < iterations; i++) {
        shared_counter++;
    }
    printf("[PTHREAD] Запущен: PID=%ld, TID=%ld\n", (long)getpid(), (long)syscall(SYS_gettid));
    return NULL;
}

int main(void) {
    pthread_t thread;
    pthread_create(&thread, NULL, worker, (void*)10000);
    pthread_join(thread, NULL);
    printf("[MAIN] Финальное значение счётчика: %ld\n", shared_counter);
    return 0;
}
```

### 3. Полный код bench.c и таблица результатов `perf stat`

#### 3.1 Исходный код bench.c
```c
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>
#include <sys/types.h>
#include <sys/syscall.h>
#include <sched.h>
#include <pthread.h>

#define STACK_SIZE (64 * 1024)

int dummy_func(void *arg) {
    (void)arg;
    volatile int c = 0;
    for (int i = 0; i < 10000; i++) c++;
    return 0;
}

void *pthread_func(void *arg) {
    dummy_func(arg);
    return NULL;
}

void run_fork(int n) {
    for (int i = 0; i < n; i++) {
        pid_t pid = fork();
        if (pid == 0) {
            dummy_func(NULL);
            exit(0);
        }
    }
    while (wait(NULL) > 0);
}

void run_pthread(int n) {
    pthread_t *threads = malloc(sizeof(pthread_t) * n);
    for (int i = 0; i < n; i++) {
        pthread_create(&threads[i], NULL, pthread_func, NULL);
    }
    for (int i = 0; i < n; i++) {
        pthread_join(threads[i], NULL);
    }
    free(threads);
}

void run_clone(int n) {
    pid_t *tids = malloc(sizeof(pid_t) * n);
    for (int i = 0; i < n; i++) {
        char *stack = malloc(STACK_SIZE);
        char *stack_top = stack + STACK_SIZE;
        pid_t tid = clone(dummy_func, stack_top, CLONE_VM|CLONE_FS|CLONE_FILES|CLONE_SIGHAND|CLONE_THREAD|SIGCHLD, NULL);
        tids[i] = tid;
    }
    // Краткое ожидание завершения легковесных задач
    unsigned int remaining = 100000;
    while (remaining > 0) remaining--;
    free(tids);
}

int main(int argc, char *argv[]) {
    if (argc < 3) return 1;
    int n = atoi(argv[2]);
    if (strcmp(argv[1], "fork") == 0) run_fork(n);
    else if (strcmp(argv[1], "pthread") == 0) run_pthread(n);
    else if (strcmp(argv[1], "clone") == 0) run_clone(n);
    return 0;
}
```

### 5. Исходный код clone_probe.c и лог dmesg трассировки kprobe

#### 5.1 Исходный код модуля ядра clone_probe.c
```c
#include <linux/module.h>
#include <linux/kprobes.h>
#include <linux/sched.h>

MODULE_LICENSE("GPL");
MODULE_AUTHOR("OS Student");
MODULE_DESCRIPTION("Kprobe _do_fork tracer");
MODULE_VERSION("1.0");

static struct kprobe kp = {
    .symbol_name = "_do_fork", // Адаптировано под ядро QEMU 4.19
};

static int handler_pre(struct kprobe *p, struct pt_regs *regs) {
    printk(KERN_INFO "clone_probe: [DETECTED] pid=%d tgid=%d comm=%s\n",
           current->pid, current->tgid, current->comm);
    return 0;
}

static int __init kprobe_init(void) {
    kp.pre_handler = handler_pre;
    int ret = register_kprobe(&kp);
    if (ret < 0) {
        printk(KERN_ERR "clone_probe: register_kprobe failed, returned %d\n", ret);
        return ret;
    }
    printk(KERN_INFO "clone_probe: kprobe registered successfully on _do_fork\n");
    return 0;
}

static void __exit kprobe_exit(void) {
    unregister_kprobe(&kp);
    printk(KERN_INFO "clone_probe: kprobe unregistered\n");
}

module_init(kprobe_init);
module_exit(kprobe_exit);
```

#### 6.1 Изменённый фрагмент minishell.c (Реализация pthread-конвейера через общий буфер)
```c
// Чтение байта из кольцевого буфера в общем пространстве памяти с условными переменными
int buffer_get(SharedBuffer *buf, char *c) {
    pthread_mutex_lock(&buf->lock);
    while (buf->count == 0) {
        if (buf->is_eof) {
            pthread_mutex_unlock(&buf->lock);
            return 0; 
        }
        pthread_cond_wait(&buf->not_empty, &buf->lock);
    }
    *c = buf->data[buf->head];
    buf->head = (buf->head + 1) % BUFFER_SIZE;
    buf->count--;
    pthread_cond_signal(&buf->not_full);
    pthread_mutex_unlock(&buf->lock);
    return 1;
}
```

### 7. Ответы на контрольные вопросы
1. Чем процесс отличается от потока исполнения? Что у них общего с точки зрения ядра Linux?
·	У процесса может быть несколько потоков. Потоки — это, грубо говоря, подзадачи основной цели (процесса). Также процесс обладает своей обособленной памятью, ничего не знает о других процессах и думает, что вся память компьютера принадлежит ему.
·	Процесс и поток ядро Linux видит одинаково — как задачи (tasks) и не различает их при планировании. Различать их внутри этой ОС можно по структуре памяти: если дескрипторы памяти разные — это независимые процессы, а если структура памяти одна на двоих — это потоки одного процесса.
2. Что представляет собой task_struct (аналог PCB в Linux)? Какие группы данных в нём отражают состояние планирования, адресное пространство, файловые дескрипторы, сигналы, идентификаторы, cgroup и другие ресурсы задачи?
·	task_struct — это описатель или структура задачи (процесса или потока) в памяти ядра. В зависимости от того, что перед нами, её поля указывают на разные разделяемые ресурсы.
·	Данные внутри неё делятся на группы: за состояние планирования отвечает поле state, за адресное пространство (виртуальную память) — структура mm_struct, за открытые файлы — таблица файловых дескрипторов files_struct, за обработку сигналов — структура sighand_struct, а за системные номера — идентификаторы pid (для потоков) и tgid (для процессов).
3. Как в Linux соотносятся процесс, поток, TID и TGID? Что возвращают getpid() и gettid() в различных потоках одного процесса? Является ли pthread_t идентификатором PID или TID?
·	У процесса может быть от одного и более потоков. TID — это Thread ID, то есть системный идентификатор конкретного потока в ядре. TGID — это Thread Group ID, то есть идентификатор всей группы потоков, который для пользователя равен привычному PID (Process ID).
·	У главного потока (родителя) его личный TID равен общему TGID. У всех остальных дочерних потоков этой группы TGID одинаковый, а TID у каждого свой. Соответственно, функция getpid() всегда вернёт PID (он же TGID), а функция gettid() — уникальный системный айди конкретного потока.
·	Тип pthread_t — это внутренний идентификатор потока из библиотеки glibc (Userspace). Он работает только на уровне пользователя и может быть представлен в памяти как адрес структуры. Если попробовать найти поток по этому pthread_t внутри ядра (Kernel space), мы его там не найдём, так как ядро о нём ничего не знает.
4. Какие ресурсы могут разделяться несколькими task_struct? В частности, как разделяются адресное пространство, таблица файловых дескрипторов, обработчики сигналов и другие объекты ядра?
·	Несколько задач task_struct могут совместно использовать ресурсы под управлением системного вызова clone(). Разделяться могут: виртуальная память, таблица файловых дескрипторов (список открытых файлов), информация о текущих путях файловой системы и реакция на различные сигналы прерывания.
·	Они разделяются за счёт того, что дочерние структуры task_struct при создании копируют в свои поля прямые указатели на уже существующие структуры родительского процесса, не создавая новые копии в памяти ядра.
5. Какие области памяти являются общими у потоков одного процесса, а какие — индивидуальными? Почему у каждого потока должен быть собственный пользовательский и ядерный стек?
·	У потоков одного процесса общими являются глобальные и статические переменные, Куча (Heap для malloc), а также сегмент кода программы. Индивидуальными являются регистры процессора, локальная память потока (TLS) и стеки.
·	Каждый поток должен иметь собственный пользовательский и ядерный стек для того, чтобы во время параллельной работы не было проблем с гонкой данных (Data Race). Если бы стек был общим, то один поток мог бы вызвать свою функцию и перезаписать данные стека соседа (а в стеке хранятся данные об адресе возврата и локальные переменные). Из-за этого программа прочитала бы неправильные данные для завершения работы функции, что вызвало бы аппаратный крах — Segmentation fault.
6. Как устроена модель потоков 1:1 в Linux/NPTL? Почему каждому POSIX-потоку обычно соответствует отдельная задача ядра и собственный task_struct?
·	В Linux используется модель 1:1. Это означает, что одному потоку из пользовательского пространства (Userspace) строго соответствует один поток из ядерного (Kernelspace).
·	Главный плюс такой модели — если один поток зависнет или заблокируется на чтении файла, другие потоки продолжат работать на своих ядрах CPU. Главный минус — для создания каждого потока требуется делать системный вызов к ядру, что является более долгой и дорогой операцией по сравнению с работой в пространстве пользователя. Каждому потоку нужна своя task_struct, чтобы планировщик ядра мог распределять их по разным физическим процессорам.
7. Как создаётся и завершается POSIX-поток? Назначение pthread_create(), pthread_join(), joinable- и detached-потоков, а также особенности обработки кодов ошибок и назначение опции компилятора -pthread.
·	Поток создаётся функцией pthread_create(), а завершается самостоятельно (выходом из функции или через pthread_exit). Функция pthread_join() нужна для того, чтобы дождаться завершения потока и очистить его память (обычные вызовы ожидания процессов wait() здесь не сработают).
·	Joinable-поток — это поток по умолчанию; система ожидает, пока мы вызовем для него pthread_join(), так как он важен и может вернуть результат работы. Detached-поток — это отсоединённый поток, его результат нам не важен, и сразу после его смерти система автоматически удаляет остатки его ресурсов из памяти.
·	Особенность обработки ошибок в pthreads заключается в том, что функции возвращают 0 (если всё отлично) или прямой код ошибки (например, EINVAL), поэтому в коде мы проверяем условие != 0. Опция компилятора -pthread необходима для подключения библиотек и настройки функций стандартной библиотеки Си в потокобезопасный режим.
8. Чем pthread_join() отличается от wait() и waitpid()? Какие объекты и какие состояния завершения они синхронизируют?
·	Отличаются они тем, что функция pthread_join() работает строго с POSIX-потоками на уровне пользователя и ждёт конкретный поток по его pthread_t. Системные вызовы wait() и waitpid() работают с процессами на уровне ядра и не могут ожидать отдельные потоки внутри одной группы.
·	Они синхронизируют разные состояния: wait переводит родительский процесс в ожидание изменения состояния дочернего процесса (его смерти), а pthread_join усыпляет поток на специальном механизме futex в ожидании момента, когда целевой поток завершится и обнулит свой TID.
9. Как clone() формирует границу между «процессом» и «потоком» в Linux? Раскройте назначение как минимум CLONE_VM, CLONE_FILES, CLONE_THREAD, CLONE_SIGHAND, CLONE_FS, CLONE_PARENT_SETTID, CLONE_CHILD_CLEARTID.
·	Если мы вызываем clone() без флагов — мы клонируем полноценный изолированный дочерний процесс. Если мы добавляем специальные флаги разделения ресурсов — мы конструируем поток.
·	CLONE_VM (Virtual Memory) — создаёт задачу с общим виртуальным адресным пространством (общая память).
·	CLONE_FILES — задаёт общую таблицу файловых дескрипторов (какие файлы открыты, с теми одновременно могут работать все потоки).
·	CLONE_FS (File System) — делит информацию о файловой системе (текущая рабочая директория cwd становится общей).
·	CLONE_THREAD — копирует новую задачу в текущую группу потоков родителя, присваивая ей общий TGID (общий PID).
·	CLONE_SIGHAND (Signal Handlers) — задаёт общую обработку сигналов (вся группа потоков будет иметь одинаковую, в том числе кастомную, реакцию на системные сигналы).
·	CLONE_PARENT_SETTID — приказывает ядру записать системный TID нового потока в память родителя прямо в момент создания, чтобы избежать гонки данных при запуске.
·	CLONE_CHILD_CLEARTID — заставляет ядро автоматически очистить (обнулить) TID дочернего потока в памяти и послать сигнал пробуждения в момент его смерти, что позволяет работать функции pthread_join().
10. Почему комбинация CLONE_THREAD без CLONE_VM недопустима? Почему набора CLONE_VM | CLONE_THREAD недостаточно для самостоятельной полноценной реализации POSIX-потоков?
·	Потому что по логике операционной системы у задач, входящих в одну группу потоков (CLONE_THREAD), обязана быть общая виртуальная память. Иначе они не смогут работать с глобальными данными.
·	Набора только этих двух флагов недостаточно, так как для полноценной работы POSIX-потоков нам нужно как минимум включить одинаковую обработку сигналов (CLONE_SIGHAND), настроить локальное хранилище потока (TLS) для изоляции переменной ошибок errno и выделить под каждый поток собственный безопасный стек.
11. Зачем GNU-обёртке clone() нужен отдельный стек? Чем пользовательский стек потока отличается от стека ядра?
·	Так как при вызове clone с флагом CLONE_VM память становится общей, дочернему потоку обязательно нужен свой собственный кусок памяти под стек. Если его не выделить, дочерний поток начнет использовать тот же стек, что и родитель, затирая его данные и ломая адреса возврата функций.
·	Пользовательский стек выделяется в обычной памяти программы (Userspace) под локальные переменные функций. Стек ядра — это маленький фиксированный буфер (обычно 8-16 КБ) в памяти самого ядра, который создается для каждой task_struct и используется процессором только тогда, когда поток выполняет системный вызов.
12. Как CLONE_CHILD_CLEARTID и futex используются для ожидания завершения потока? Как это связано с внутренней реализацией pthread_join()?
·	Функция pthread_join() заставляет вызывающий поток уснуть на адресе памяти умирающего потока с помощью системного вызова futex (быстрый пользовательский мьютекс).
·	Благодаря флагу CLONE_CHILD_CLEARTID, ядро Linux перед самым удалением потока само записывает 0 по этому адресу памяти и вызывает команду FUTEX_WAKE. Этот вызов будит спящего родителя, сообщая ему, что поток завершился.
13. Какая гонка возникает при параллельном выполнении counter++ несколькими потоками? Почему общая память сама по себе не обеспечивает атомарность операции и какие механизмы синхронизации применяются?
·	Чтобы увеличить значение переменной на уровне процессора, программа выполняет три отдельных действия: считать значение из кучи (Heap) в регистр, увеличить его на 1 внутри регистра процессора и записать обратно в память.
·	Потоки, которые не разграничены блокировками, могут одновременно считать из памяти, например, число 10. Каждый у себя увеличит его до 11 и запишет обратно. В итоге вместо 12 в памяти останется 11. Это называется гонкой данных. Общая память просто дает доступ к байтам, но не делает операцию неделимой (атомарной). Для защиты применяются мьютексы (pthread_mutex_t), которые закрывают доступ к переменной для других потоков на время расчета.
14. Чем отличаются signal disposition и signal mask в многопоточной программе? Как доставляются сигналы процессу и конкретному потоку, чем синхронные сигналы отличаются от асинхронных?
·	Signal disposition — это общее для всего процесса правило (диспозиция), определяющее, как программа реагирует на сигнал (например, завершиться по Ctrl+C). Signal mask — это индивидуальная маска-щит конкретного потока, позволяющая ему временно заблокировать получение определенных сигналов лично для себя.
·	Синхронные сигналы (вызванные ошибками самого потока, например, делением на ноль) доставляются строго в поток-виновник. Асинхронные сигналы (внешние, от других программ) доставляются на уровень процесса, и ядро выбирает любой случайный поток, у которого этот сигнал не закрыт маской.
15. Что происходит с остальными потоками процесса при выполнении exec() одним из потоков? Как при exec() заменяется адресное пространство и что происходит с файловыми дескрипторами, помеченными close-on-exec?
·	Если один из потоков вызывает exec(), текущий образ программы полностью стирается. Все остальные потоки этого процесса принудительно уничтожаются ядром Linux.
·	Адресное пространство полностью заменяется кодом новой программы с диска. Файловые дескрипторы, которые при открытии были помечены флагом O_CLOEXEC (close-on-exec), автоматически закрываются ядром, а остальные — передаются новой программе.
16. Почему fork() в многопоточной программе требует особой осторожности? Какие проблемы возможны с блокировками, состоянием библиотек и вызовами до последующего exec()?
·	fork() — это копирование процесса, при котором в дочерний процесс копируется только один поток (вызыватель), а все остальные потоки исчезают.
·	Может быть такое, что в родительском процессе какой-то другой поток закрыл замок (мьютекс) на общую переменную и в этот момент умер (не скопировался в новый процесс). Из-за этого в созданном дочернем процессе этот мьютекс останется закрытым навсегда, так как освободить его больше некому. Единственный поток дочернего процесса зависнет при попытке обратиться к этой памяти. Поэтому до вызова exec() нельзя использовать сложные функции библиотек.
17. Что делают системные вызовы fork(), exec(), wait()/waitpid() и exit()? Почему в традиционной Unix-модели запуск программы разделён на fork() и exec(), а не выражен одним системным вызовом spawn()?
·	fork() — создание дочернего процесса; exec() — замена текущего процесса новой программой; wait()/waitpid() — ожидание завершения процесса родителем; exit() — завершение текущей задачи с освобождением ресурсов.
·	Разделение на два вызова вместо одного spawn нужно для того, чтобы в промежутке между копированием (fork) и запуском новой программы (exec) программист мог гибко настроить окружение внутри дочернего процесса: например, закрыть лишние файлы или перенаправить каналы ввода-вывода (как при работе конвейера пайпов).
18. Что происходит в ядре при fork()? Какие структуры копируются, какие первоначально разделяются, как наследуются файловые дескрипторы и сигнальные диспозиции?
·	Ядро создает новую карточку задачи task_struct, копирует дескриптор памяти mm_struct и обработчики сигналов. Таблица файловых дескрипторов дублируется как массив: потомок получает те же индексы файлов, а сами открытые файлы внутри ядра начинают разделяться (увеличивается их счетчик ссылок). Сигнальные диспозиции полностью наследуются от родителя.
19. Как реализован copy-on-write (COW) при fork()? Что происходит с записями таблиц страниц, правами доступа и счётчиками ссылок, когда один из процессов впервые записывает в ранее общую страницу?
·	При fork() ядро не копирует физическую память, а просто копирует таблицы страниц и помечает всю память родителя и потомка как Read-Only (только для чтения).
·	Когда один из процессов пытается записать данные в такую страницу, процессор выдает аппаратную ошибку страницы (Page Fault). Ядро перехватывает её, физически копирует эту конкретную страницу памяти в новое место ОЗУ, меняет для пишущего процесса права доступа на Read-Write и уменьшает счетчик ссылок на старую общую страницу.
20. Почему изменение глобальной переменной потомком после fork() обычно не видно родительскому процессу? Почему одинаковый виртуальный адрес в двух процессах не доказывает наличие общей физической памяти?
·	Потому что у них логически разные адресные пространства. В момент записи срабатывает механизм COW, и переменная потомка физически переносится в другую ячейку памяти.
·	Одинаковый виртуальный адрес не доказывает общую память, так как программы видят только абстрактные цифры адресов, а блок процессора MMU переводит один и тот же виртуальный адрес в совершенно разные физические адреса на плашках оперативной памяти по таблицам страниц конкретного процесса.
21. Что происходит при завершении процесса? Чем zombie-процесс отличается от orphan-процесса, почему требуется wait()/waitpid() и как ядро окончательно освобождает ресурсы потомка?
·	При завершении процесса умирают его потоки, закрываются файлы и освобождается память. Но бывает так, что процесс уже завершился, а его карточка задачи task_struct все еще хранится в таблице процессов ядра, так как родитель еще не прочитал его код возврата — это zombie-процесс (зомби).
·	Orphan-процесс (сирота) — это работающий процесс, у которого умер родитель; его автоматически усыновляет процесс init (PID 1). Вызовы wait() необходимы, чтобы считать код возврата и дать ядру команду окончательно стереть структуры умершего потомка из памяти.
22. Какие существуют состояния процесса или потока? Опишите состояния running, ready/runnable, blocked/waiting, stopped и zombie, а также типичные переходы между ними.
·	Running — задача выполняется прямо сейчас на ядре CPU.
·	Ready / Runnable — задача исправна и готова к работе, но ждет своей очереди в планировщике.
·	Blocked / Waiting — задача «спит», ожидая внешнего события (например, ввода с клавиатуры или диска).
·	Stopped — поток остановлен управляющим сигналом (например, по Ctrl+Z).
·	Zombie — процесс завершил работу, но запись о нем еще хранится в таблице задач.
·	Переходы: Из состояния сна (Blocked) задача переходит в готовность (Ready), планировщик берет её в работу (Running), при ожидании ввода она снова засыпает (Blocked), а при выходе — превращается в Zombie.
23. Что такое PID, PPID, PGID и session ID? Чем процесс-группа отличается от сессии и для каких задач используются эти иерархии?
·	PID — Process ID (номер текущего процесса). PPID — Parent Process ID (номер родительского процесса). PGID — Process Group ID (идентификатор группы процессов, связанных, например, в один конвейер пайпов). Session ID — идентификатор сессии, объединяющей группы процессов под управлением одного терминала.
·	Группа процессов нужна для одновременной отправки сигналов (например, Ctrl+C убивает сразу весь конвейер), а сессия разграничивает активные и фоновые задачи терминала.
24. Как ядро изолирует адресные пространства разных процессов? Какую роль играют таблицы страниц, права доступа, режимы процессора и переключение активного адресного пространства?
·	Ядро изолирует память с помощью таблиц страниц и разделения режимов работы процессора: код пользователя работает в ограниченном режиме Ring 3, а ядро — в привилегированном Ring 0.
·	При создании процесса ядро выделяет ему персональную древовидную таблицу страниц. При переключении на новый процесс ядро записывает адрес его таблицы страниц в специальный регистр процессора CR3. Процессор аппаратно не может прочитать чужую память, так как видит адреса только через таблицу, заданную регистром CR3.
25. Как происходит переключение контекста на x86-64? Какие компоненты состояния потока необходимо сохранить и восстановить, как используется отдельный стек ядра и почему переключение выполняется между задачами, а не между абстрактными «процессами»?
·	При смене контекста ядро сохраняет текущие значения регистров процессора общего назначения, указатель текущей команды (RIP) и указатель стека (RSP) в структуру уходящей задачи, а затем восстанавливает эти же значения из структуры новой задачи.
·	У каждого потока есть свой изолированный стек ядра, адрес которого при переключении прописывается в системный сегмент TSS, чтобы прерывания нового потока обрабатывались в его личном пространстве. Переключение происходит между задачами (task_struct), так как для процессора и планировщика потоки и процессы с точки зрения смены регистров абсолютно равноправны.
26. Чем потоки ядра отличаются от пользовательских потоков (green threads)? В чём состоят достоинства и ограничения моделей M:1, 1:1 и M:N?
·	Потоки ядра (task_struct) управляются планировщиком операционной системы и могут выполняться на разных физических ядрах CPU. Пользовательские потоки (green threads) создаются внутри самой программы, ядро о них ничего не знает, а их переключением занимается планировщик языка программирования.
·	Модель 1:1 (Linux): Одному пользовательскому потоку соответствует один ядерный. Плюс: Настоящий параллелизм на многоядерных процессорах, зависание одного потока не вешает другие. Минус: Дорого и долго по времени создавать потоки через системные вызовы ядра.
·	Модель M:1: Множество потоков программы крутятся внутри одного потока ядра. Плюс: Очень быстрое переключение без системных вызовов. Минус: Не могут использовать многоядерность, а при блокировке одного потока зависнет вся программа.
·	Модель M:N: Гибридная модель (например, в Go). Большое число потоков программы распределяется силами рантайма поверх фиксированного пула потоков ядра. Объединяет скорость модели M:1 и параллелизм модели 1:1.
